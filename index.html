Configura tus credenciales de Airtable const AIRTABLE_BASE_ID = 'tu_base_id_aqui'; const AIRTABLE_TABLE_NAME = 'Propiedades'; const AIRTABLE_API_TOKEN = 'tu_personal_access_token_aqui'; // Usa Personal Access Tokens (PAT) const url = `https://api.airtable.com/v0/${AIRTABLE_BASE_ID}/${AIRTABLE_TABLE_NAME}`; // 2. Mapeo de colores según el estatus (Formatos HEX con opacidad si lo soporta tu API) const coloresEstatus = { "Disponible": "#2ecc71", // Verde "Apartado": "#f1c40f", // Amarillo "Vendido": "#e74c3c" // Rojo }; // 3. Función para consultar Airtable y actualizar el tour async function actualizarTourDesdeAirtable() { try { const respuesta = await fetch(url, { headers: { Authorization: `Bearer ${AIRTABLE_API_TOKEN}` } }); const datos = await respuesta.json(); // Recorremos cada registro devuelto por Airtable datos.records.forEach(record => { const idHotspot = record.fields.ID_Hotspot; const estatus = record.fields.Estatus; const precio = record.fields.Precio; // Buscamos el color correspondiente al estatus const colorAsignado = coloresEstatus[estatus] || "#ffffff"; // 4. Le ordenamos a 3DVista que cambie el color del polígono // Nota: Si usas la API nativa o Blaze IT, la función exacta puede variar ligeramente. // Ejemplo conceptual usando la lógica de Blaze IT: const hotspotElemento = getHotspotByName(idHotspot); if (hotspotElemento) { setHotspotColor(hotspotElemento, colorAsignado); // Opcional: Si quieres que al hacer clic muestre los datos de Airtable hotspotElemento.onClick = function() { mostrarPopupPersonalizado(idHotspot, estatus, precio); }; } }); } catch (error) { console.error("Error al conectar con Airtable:", error); } } // 5. Ejecutar la función cuando el tour virtual esté completamente cargado window.on3DVistaLoad = function() { actualizarTourDesdeAirtable(); };
